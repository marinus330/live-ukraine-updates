  <div id="updates">Loading updates...</div>

  <script>
    async function fetchUpdates() {
      const proxy = "https://corsproxy.io/?";
      const feedUrl = "https://liveuamap.com/rss";
      const fullUrl = proxy + encodeURIComponent(feedUrl);

      try {
        const response = await fetch(fullUrl);
        const text = await response.text();
        const parser = new DOMParser();
        const xml = parser.parseFromString(text, "text/xml");
        const items = xml.querySelectorAll("item");

        const container = document.getElementById("updates");
        container.innerHTML = "";

        items.forEach((item, i) => {
          if (i >= 10) return;
          const title = item.querySelector("title").textContent;
          const date = item.querySelector("pubDate").textContent;

          container.innerHTML += `
            <div class="update">
              <strong>${title}</strong><br/>
              <div class="timestamp">${date}</div>
            </div>
          `;
        });

      } catch (err) {
        document.getElementById("updates").innerText = "Failed to load updates.";
        console.error("Fetch error:", err);
      }
    }

    fetchUpdates();
    setInterval(fetchUpdates, 60000); // Refresh every 60 seconds
  </script>
